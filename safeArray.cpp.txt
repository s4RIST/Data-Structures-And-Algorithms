int safeArray::pushback(int val) {
    if (size == capacity) {
        cout << "\nPushback\n";
        int* temp = arr;
        arr = new int[size * 2];

        for (int i = 0; i < size; i++) {
            arr[i] = temp[i];
        }
        for (int i = size; i < size * 2; i++) {
            arr[i] = 0;
        }
        delete[] temp;
        temp = nullptr;
        capacity = capacity* 2; 

        if (capacity > size) {
            arr[size] = val;
            size++;
        }
        return 0;
    }
}

safeArray::safeArray(const safeArray& array) {
    cout << "\nCopy Constructor Called\n";
    size = array.size;
    capacity = array.capacity;
    arr = new int[capacity];

    for (int i = 0; i < capacity; i++)
        arr[i] = array.arr[i];
}


safeArray& safeArray::operator=(const safeArray& rhs)
{
    delete[] arr;
    cout << "\nCopy Assignment Called\n";
    size = rhs.size;
    capacity = rhs.capacity;
    arr = new int[size]; //new array with pointer of size

    if (this != &rhs)
    {
        for (int i = 0; i < size; i++) {
            arr[i] = rhs.arr[i];
        }
    }
    return *this;
    cout << "In the Copy Assignment Operator" << endl;
}
