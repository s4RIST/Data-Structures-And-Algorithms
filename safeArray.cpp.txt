        }
    }

        for (int i = 0; i < size; i++) //loop over the array to insert values.

            arr[i] = i;

        }
}

}

ostream& operator<<(ostream& lhs, const safeArray& rhs) {
    for (int i = 0; i < rhs.size; i++) {
        lhs << rhs.arr[i];
    }
    return lhs;
}
