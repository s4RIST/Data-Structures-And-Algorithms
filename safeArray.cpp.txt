#include <iostream>
using namespace std;
#include "safeArray.h"

safeArray::safeArray()
{
}

safeArray::safeArray(int size)
{
    arr = new int[size];
    for (int i = 0; i < size; i++) //loop over the array to insert values.
    {
        arr[i] = i;
        cout << i;
    }
}

int safeArray::at(int p)
{
    return arr[p];
}

int safeArray::add(int position, int element) 
{
    arr[position] = element;
}


safeArray::~safeArray()
{
    delete[] arr;
    arr = nullptr;  // (>>>>RAW POINTER<<<< IS POINTING AT NOTHING)

    cout << "Destructor called" << endl;
}
